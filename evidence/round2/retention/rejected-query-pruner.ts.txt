// REJECTED policy, retained as the session's recorded source checkpoint.
// It is not imported by the application. The bug is synchronous pruning during
// observerRemoved: a newly built pending query has zero observers until the
// observer finishes moving from the old query, so the callback can delete it.
const MAX_INACTIVE_RUN_VIEWS = 12;
let pruning = false;
client.getQueryCache().subscribe((event) => {
  if (pruning || (event.type !== 'observerRemoved' && event.type !== 'updated')) return;
  pruning = true;
  try {
    const inactive = client.getQueryCache().findAll({ queryKey: ['run'] })
      .filter((query) => query.getObserversCount() === 0 && query.state.fetchStatus === 'idle')
      .sort((a, b) => a.state.dataUpdatedAt - b.state.dataUpdatedAt);
    for (const query of inactive.slice(0, Math.max(0, inactive.length - MAX_INACTIVE_RUN_VIEWS))) {
      client.getQueryCache().remove(query);
    }
  } finally { pruning = false; }
});
